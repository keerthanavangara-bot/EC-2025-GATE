\end{align}

p_X\sbrak{2}&=p_B\sbrak{1}p_B\sbrak{1}=\frac{1}{4}

\begin{align}

&=\frac{1}{16}+\frac{1}{4}+\frac{1}{16}

&=\frac{3}{8}
\end{align}
